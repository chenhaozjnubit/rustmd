use std::collections::HashMap;

/// 计算斐波那契数列
fn fib(n: u64) -> u64 {
    let mut memo: HashMap<u64, u64> = HashMap::new();
    (0..=n).fold((0, 1), |(a, b), i| {
        memo.insert(i, a);
        (b, a + b)
    }).1
}
